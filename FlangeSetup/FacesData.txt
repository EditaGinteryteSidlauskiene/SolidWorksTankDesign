//! # Table 8 (Flange Facing Dimensions)
//!
//! **Identity**:
//! This module implements Table 8 from EN 1092-1, providing dimensions for various flange facing types.
//!
//! **Functionality**:
//! It retrieves the specific dimensions required for a given `FaceType`, dependent on the flange's
//! Nominal Diameter (DN) and Nominal Pressure (PN). It strictly enforces the availability of face types
//! for specific PN ranges (e.g., Types G and H are only available for PN 10 to PN 40).
//!
//! **Composition**:
//! *   `FaceDimensions`: An enum with variants for each face type, holding the specific dimensions for that face.
//! *   `get_face_dimensions`: The primary lookup function.
//! *   Internal helper functions (implied) to structure the data lookup.
//!
//! **Usage**:
//! This module is used to obtain the precise geometry of the flange face for rendering or specification.
//!
//! ```rust
//! use en1092_1::enums::{Dn, FaceType, Pn};
//! use en1092_1::tables::face_dimensions::{get_face_dimensions, FaceDimensions};
//!
//! if let Some(FaceDimensions::TypeB { d1, f1 }) = get_face_dimensions(FaceType::TypeB, Dn::Dn40, Pn::Pn16) {
//!     assert_eq!(d1, 88.0);
//!     assert_eq!(f1, 3.0);
//! }
//! ```

use crate::enums::{Dn, FaceType, Pn};

/// Represents the set of dimensions required for a specific face type.
#[derive(Debug, Clone, Copy, PartialEq)]
pub enum FaceDimensions {
    /// Type A: Flat face. No specific dimensions required beyond the flange thickness.
    TypeA,
    /// Type B: Raised face (B1 or B2).
    TypeB {
        /// Diameter of raised face (`d1`).
        d1: f64,
        /// Height of raised face (`f1`).
        f1: f64,
    },
    /// Type C: Tongue.
    TypeC {
        /// Height of tongue (`f2`).
        f2: f64,
        /// Outside diameter of tongue (`w`).
        w: f64,
        /// Inside diameter of tongue (`x`).
        x: f64,
    },
    /// Type D: Groove.
    TypeD {
        /// Diameter of groove (`d1`).
        d1: f64,
        /// Depth of groove (`f1`).
        f1: f64,
        /// Outside diameter of groove (`y`).
        y: f64,
        /// Inside diameter of groove (`z`).
        z: f64,
        /// Depth of groove (`f3`).
        f3: f64,
    },
    /// Type E: Spigot.
    TypeE {
        /// Height of spigot (`f2`).
        f2: f64,
        /// Diameter of spigot (`x`).
        x: f64,
    },
    /// Type F: Recess.
    TypeF {
        /// Diameter of recess (`d1`).
        d1: f64,
        /// Depth of recess (`f1`).
        f1: f64,
        /// Diameter of recess (`y`).
        y: f64,
        /// Depth of recess (`f3`).
        f3: f64,
    },
    /// Type G: O-ring recess.
    TypeG {
        /// Diameter of recess (`d1`).
        d1: f64,
        /// Depth of recess (`f1`).
        f1: f64,
        /// Diameter of recess (`w`).
        w: f64,
    },
    /// Type H: O-ring groove.
    TypeH {
        /// Depth of groove (`f3`).
        f3: f64,
        /// Depth of groove (`f4`).
        f4: f64,
        /// Diameter of groove (`y`).
        y: f64,
        /// Diameter of groove (`z`).
        z: f64,
        /// Angle of groove (`alpha`).
        alpha: f64,
        /// Radius of groove (`r`).
        r: f64,
    },
}

/// Retrieves the face dimensions for a given face type, DN, and PN.
///
/// # Arguments
///
/// * `face_type` - The type of flange face (A-H).
/// * `dn` - The nominal diameter (DN).
/// * `pn` - The nominal pressure (PN).
///
/// # Returns
///
/// Returns `Some(FaceDimensions)` containing the specific dimensions if the combination is valid.
/// Returns `None` if the face type is not applicable for the given PN (e.g., Type C for PN 6).
///
/// # Remarks
///
/// This function implements **Table 8** of EN 1092-1:2018.
///
/// *   Dimensions return `Some` only if the face type is standard for the given DN/PN.
/// *   The field names (`d1`, `f1`, `w`, etc.) correspond strictly to the column headers in the standard text.
pub const fn get_face_dimensions(face_type: FaceType, dn: Dn, pn: Pn) -> Option<FaceDimensions> {
    // -------------------------------------------------------------------------
    // 1. Validate PN Restrictions (EN 1092-1:2018 Table 8 footnotes)
    // -------------------------------------------------------------------------
    match face_type {
        // "Flange facing types C, D, E, F, G and H ... are not used for PN 2,5 and 6."
        FaceType::TypeC
        | FaceType::TypeD
        | FaceType::TypeE
        | FaceType::TypeF
        | FaceType::TypeG
        | FaceType::TypeH => match pn {
            Pn::Pn2_5 | Pn::Pn6 => return None,
            _ => {}
        },
        _ => {}
    }

    match face_type {
        // "Flange facing types G and H ... are only used for PN 10 to PN 40."
        FaceType::TypeG | FaceType::TypeH => match pn {
            Pn::Pn10 | Pn::Pn16 | Pn::Pn25 | Pn::Pn40 => {}
            _ => return None,
        },
        _ => {}
    }

    // -------------------------------------------------------------------------
    // 2. Retrieve Raw Values & Construct Enum Variant
    // -------------------------------------------------------------------------
    let dn_val = dn.value();

    match face_type {
        FaceType::TypeA => Some(FaceDimensions::TypeA),
        FaceType::TypeB => {
            if let (Some(d1), Some(f1)) = (get_d1(dn_val, pn), get_f1(dn_val)) {
                Some(FaceDimensions::TypeB { d1, f1 })
            } else {
                None
            }
        }
        FaceType::TypeC => {
            if let (Some(f2), Some(w), Some(x)) = (get_f2(dn_val), get_w(dn_val), get_x(dn_val)) {
                Some(FaceDimensions::TypeC { f2, w, x })
            } else {
                None
            }
        }
        FaceType::TypeD => {
            if let (Some(d1), Some(f1), Some(y), Some(z), Some(f3)) = (
                get_d1(dn_val, pn),
                get_f1(dn_val),
                get_y(dn_val),
                get_z(dn_val),
                get_f3(dn_val),
            ) {
                Some(FaceDimensions::TypeD { d1, f1, y, z, f3 })
            } else {
                None
            }
        }
        FaceType::TypeE => {
            if let (Some(f2), Some(x)) = (get_f2(dn_val), get_x(dn_val)) {
                Some(FaceDimensions::TypeE { f2, x })
            } else {
                None
            }
        }
        FaceType::TypeF => {
            if let (Some(d1), Some(f1), Some(y), Some(f3)) = (
                get_d1(dn_val, pn),
                get_f1(dn_val),
                get_y(dn_val),
                get_f3(dn_val),
            ) {
                Some(FaceDimensions::TypeF { d1, f1, y, f3 })
            } else {
                None
            }
        }
        FaceType::TypeG => {
            if let (Some(d1), Some(f1), Some(w)) =
                (get_d1(dn_val, pn), get_f1(dn_val), get_w(dn_val))
            {
                Some(FaceDimensions::TypeG { d1, f1, w })
            } else {
                None
            }
        }
        FaceType::TypeH => {
            if let (Some(f3), Some(f4), Some(y), Some(z), Some(alpha), Some(r)) = (
                get_f3(dn_val),
                get_f4(dn_val),
                get_y(dn_val),
                get_z(dn_val),
                get_alpha(dn_val),
                get_r(dn_val),
            ) {
                Some(FaceDimensions::TypeH {
                    f3,
                    f4,
                    y,
                    z,
                    alpha,
                    r,
                })
            } else {
                None
            }
        }
    }
}

#[inline(always)]
const fn get_f1(dn: u32) -> Option<f64> {
    if dn <= 32 {
        Some(2.0)
    } else if dn <= 250 {
        Some(3.0)
    } else if dn <= 500 {
        Some(4.0)
    }
    // 300-500
    else if dn <= 4000 {
        Some(5.0)
    }
    // 600-4000
    else {
        None
    }
}

    // Helper for parsing CSV
    struct CsvRow {
        dn: u32,
        d1_pn2_5: Option<f64>,
        d1_pn6: Option<f64>,
        d1_pn10: Option<f64>,
        d1_pn16: Option<f64>,
        d1_pn25: Option<f64>,
        d1_pn40: Option<f64>,
        d1_pn63: Option<f64>,
        d1_pn100: Option<f64>,
        d1_pn160: Option<f64>,
        d1_pn250: Option<f64>,
        d1_pn320: Option<f64>,
        d1_pn400: Option<f64>,
        f1: Option<f64>,
        f2: Option<f64>,
        f3: Option<f64>,
        f4: Option<f64>,
        w: Option<f64>,
        x: Option<f64>,
        y: Option<f64>,
        z: Option<f64>,
        alpha: Option<f64>,
        r: Option<f64>,
    }

    fn parse_csv() -> Vec<CsvRow> {
        let content = include_str!("data/table_8.csv");
        let mut rows = Vec::new();

        for (i, line) in content.lines().enumerate() {
            // Skip header
            if i == 0 {
                continue;
            }
            if line.trim().is_empty() {
                continue;
            }

            let parts: Vec<&str> = line.split(',').collect();
            if parts.is_empty() {
                continue;
            }
            let dn_str = parts[0].trim();
            if dn_str.is_empty() {
                continue;
            }

            // SKIP headers and footers (non-numeric DNs)
            if !dn_str.chars().next().map_or(false, |c| c.is_ascii_digit()) {
                continue;
            }

            let parse_opt = |s: &str| -> Option<f64> {
                let s = s.trim();
                if s.is_empty() {
                    None
                } else {
                    Some(s.parse::<f64>().expect("Invalid float"))
                }
            };

            let dn = dn_str.parse::<u32>().expect("Invalid DN");

            rows.push(CsvRow {
                dn,
                d1_pn2_5: parse_opt(parts[1]),
                d1_pn6: parse_opt(parts[2]),
                d1_pn10: parse_opt(parts[3]),
                d1_pn16: parse_opt(parts[4]),
                d1_pn25: parse_opt(parts[5]),
                d1_pn40: parse_opt(parts[6]),
                d1_pn63: parse_opt(parts[7]),
                d1_pn100: parse_opt(parts[8]),
                d1_pn160: parse_opt(parts[9]),
                d1_pn250: parse_opt(parts[10]),
                d1_pn320: parse_opt(parts[11]),
                d1_pn400: parse_opt(parts[12]),
                f1: parse_opt(parts[13]),
                f2: parse_opt(parts[14]),
                f3: parse_opt(parts[15]),
                f4: parse_opt(parts[16]),
                w: parse_opt(parts[17]),
                x: parse_opt(parts[18]),
                y: parse_opt(parts[19]),
                z: parse_opt(parts[20]),
                alpha: parse_opt(parts[21]),
                r: parse_opt(parts[22]),
            });
        }
        rows
    }

    #[test]
    fn test_table_8_compliance() {
        let rows = parse_csv();
        let mut failures = Vec::new();

        // Helper check_d1 can't be a closure easily due to borrowing, so we iterate manually.
        for row in rows {
            let dn_enum = match Dn::try_from(row.dn) {
                Ok(v) => v,
                Err(e) => panic!("CSV contains DN {} which is not in Dn enum: {}", row.dn, e),
            };

            // Check d1 for all PNs
            check_d1(dn_enum, Pn::Pn2_5, row.d1_pn2_5, &mut failures);
            check_d1(dn_enum, Pn::Pn6, row.d1_pn6, &mut failures);
            check_d1(dn_enum, Pn::Pn10, row.d1_pn10, &mut failures);
            check_d1(dn_enum, Pn::Pn16, row.d1_pn16, &mut failures);
            check_d1(dn_enum, Pn::Pn25, row.d1_pn25, &mut failures);
            check_d1(dn_enum, Pn::Pn40, row.d1_pn40, &mut failures);
            check_d1(dn_enum, Pn::Pn63, row.d1_pn63, &mut failures);
            check_d1(dn_enum, Pn::Pn100, row.d1_pn100, &mut failures);
            check_d1(dn_enum, Pn::Pn160, row.d1_pn160, &mut failures);
            check_d1(dn_enum, Pn::Pn250, row.d1_pn250, &mut failures);
            check_d1(dn_enum, Pn::Pn320, row.d1_pn320, &mut failures);
            check_d1(dn_enum, Pn::Pn400, row.d1_pn400, &mut failures);

            check_face_type_a(dn_enum, Pn::Pn10, &mut failures);

            // Check other faces against a known valid PN (e.g. PN10 or valid range)
            check_face_type_b(dn_enum, Pn::Pn10, row.d1_pn10, row.f1, &mut failures);
            check_face_type_c(dn_enum, Pn::Pn10, row.f2, row.w, row.x, &mut failures);
            check_face_type_d(
                dn_enum,
                Pn::Pn10,
                row.d1_pn10,
                row.f1,
                row.y,
                row.z,
                row.f3,
                &mut failures,
            );
            check_face_type_e(dn_enum, Pn::Pn10, row.f2, row.x, &mut failures);
            check_face_type_f(
                dn_enum,
                Pn::Pn10,
                row.d1_pn10,
                row.f1,
                row.y,
                row.f3,
                &mut failures,
            );
            check_face_type_g(dn_enum, Pn::Pn10, row.d1_pn10, row.f1, row.w, &mut failures);
            check_face_type_h(
                dn_enum,
                Pn::Pn10,
                row.f3,
                row.f4,
                row.y,
                row.z,
                row.alpha,
                row.r,
                &mut failures,
            );
        }

        if !failures.is_empty() {
            // Print truncated list
            for f in failures.iter().take(50) {
                println!("{}", f);
            }
            if failures.len() > 50 {
                println!("... and {} more failures.", failures.len() - 50);
            }
            panic!("Table 8 compliance match failed");
        }
    }

    fn check_d1(dn: Dn, pn: Pn, expected: Option<f64>, failures: &mut Vec<String>) {
        // Use TypeB as proxy for d1.
        let res = get_face_dimensions(FaceType::TypeB, dn, pn);
        if let Some(exp) = expected {
            if let Some(FaceDimensions::TypeB { d1, .. }) = res {
                if (d1 - exp).abs() > 1e-9 {
                    failures.push(format!(
                        "d1 mismatch DN{:?} PN{:?}: Expected {:?}, Got {:?}",
                        dn, pn, exp, d1
                    ));
                }
            } else {
                failures.push(format!(
                    "d1 mismatch DN{:?} PN{:?}: Expected {:?}, Got None (valid TypeB?)",
                    dn, pn, exp
                ));
            }
        } else {
            // expected None
            if let Some(FaceDimensions::TypeB { d1, .. }) = res {
                failures.push(format!(
                    "d1 mismatch DN{:?} PN{:?}: Expected None, Got {:?}",
                    dn, pn, d1
                ));
            }
        }
    }

    fn check_face_type_a(dn: Dn, pn: Pn, _failures: &mut Vec<String>) {
        let res = get_face_dimensions(FaceType::TypeA, dn, pn);
        assert!(matches!(res, Some(FaceDimensions::TypeA)));
    }

    fn check_face_type_b(
        dn: Dn,
        pn: Pn,
        d1: Option<f64>,
        f1: Option<f64>,
        failures: &mut Vec<String>,
    ) {
        let res = get_face_dimensions(FaceType::TypeB, dn, pn);
        match (res, d1, f1) {
            (Some(FaceDimensions::TypeB { d1: a, f1: b }), Some(e_d1), Some(e_f1)) => {
                if (a - e_d1).abs() > 1e-9 || (b - e_f1).abs() > 1e-9 {
                    failures.push(format!("TypeB mismatch DN{:?} PN{:?}: Expected d1={:?}, f1={:?}; Got d1={:?}, f1={:?}", dn, pn, e_d1, e_f1, a, b));
                }
            }
            (None, None, _) | (None, _, None) => {}
            (Some(_), None, _) => failures.push(format!(
                "TypeB mismatch DN{:?} PN{:?}: Expected None (missing d1); Got Some",
                dn, pn
            )),
            (None, Some(_), Some(_)) => failures.push(format!(
                "TypeB mismatch DN{:?} PN{:?}: Expected Some; Got None",
                dn, pn
            )),
            _ => {}
        }
    }
